<script>
function handleLogPaycheck(e) {
  e.preventDefault();
  if (state.sessionRole === 'GUEST') {
    alert('Guest role cannot log income.');
    return;
  }
  const hours = parseFloat(document.getElementById('income-hours').value) || 0;
  const rate = parseFloat(document.getElementById('income-rate').value) || 20.00;
  const otHours = parseFloat(document.getElementById('income-ot-hours').value) || 0;
  const note = document.getElementById('income-note').value.trim() || 'Standard Paycheck';

  const gross = (hours * rate) + (otHours * rate * 1.5);
  const net = gross * 0.72;

  const newLog = {
    id: Date.now(),
    date: new Date().toISOString().split('T')[0],
    hours: hours + otHours,
    gross: gross,
    net: net,
    note: note
  };

  state.incomeLogs.unshift(newLog);
  saveState('nexus_income', state.incomeLogs);
  renderIncome();

  // Trigger Glitch Animation
  const incomePanel = document.getElementById('income');
  if(incomePanel) {
    incomePanel.classList.add('flash');
    setTimeout(() => incomePanel.classList.remove('flash'), 400);
  }

  logTelemetry(`[INCOME LOGGED] Net Pay: $${net.toFixed(2)} | Note: ${note}`);
}

function renderIncome() {
  const historyContainer = document.getElementById('income-history-table');
  if(!historyContainer) return;
  historyContainer.innerHTML = state.incomeLogs.map(log => `
    <tr class="font-mono hover:bg-neutral-900">
      <td class="py-2.5 text-white">${log.date}</td>
      <td class="py-2.5 text-neutral-400">${log.hours} hrs</td>
      <td class="py-2.5 text-white">$${log.gross.toFixed(2)}</td>
      <td class="py-2.5 text-emerald-400 font-bold">$${log.net.toFixed(2)}</td>
      <td class="py-2.5 text-amber-400 text-[11px]">${log.note || '-'}</td>
      <td class="py-2.5 text-right"><button onclick="deleteIncomeLog(${log.id})">[X]</button></td>
    </tr>
  <`).join('');
}

// CLI Integration for 'payroll' and 'income'

      ).join('\n');
  }
  break;
